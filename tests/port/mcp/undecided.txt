# Attributes as another writer left them, not in normal form: two #[uses] attributes of one definition, on lines of their own and on the line of another attribute, and a definition that starts after an attribute on its line. The rewrite keeps the first, removes the others and puts a new one where the definition starts.

#[summary("Attributes written by hand")]
mod attrs {
    spec fn lo(x: i64) -> bool { x > 0 }

    spec fn hi(x: i64) -> bool { x < 9 }

    spec fn mid(x: i64) -> bool { lo(x) && hi(x) }

    spec fn top(x: i64) -> bool { mid(x) && x != 7 }

    #[effects(pure)] #[uses(lo)]
    #[uses(hi)]
    fn five() -> (r: i64)
        ensures top(r),
    {
        5
    }

    #[uses(hi)] #[uses(lo)]   #[effects(pure)]
    fn six() -> (r: i64) ensures top(r), { 6 }

    #[effects(pure)] fn four() -> (r: i64) ensures mid(r), { 4 }

    #[effects(pure)]
	#[uses(hi)]	
    fn three() -> (r: i64) ensures mid(r), { 3 }
}
start hand --undecided-info
