#[summary("Scalar parameters: each gets a slot and is read and returned, or left unused. A parameter of the unit value is not passed")]
mod params {
    #[effects(pure)]
    fn id(x: i64) -> i64 { x }
    #[effects(pure)]
    fn second(x: i64, y: bool, z: f64) -> bool { y }
    #[effects(pure)]
    fn third(x: u8, y: bool, z: f64) -> f64 { z }
    #[effects(pure)]
    fn unused(x: i32, y: u16) -> () { }
    #[effects(pure)]
    fn unit_in(u: (), x: char) -> char { x }
    #[effects(pure)]
    fn unit_out(u: ()) -> () { u }
    #[effects(pure)]
    fn konst(x: usize) -> usize { 7 }
}
