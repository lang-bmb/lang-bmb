#[summary("System interface: command-line arguments (std::env::args), environment variables (a missing name and a name containing NUL), the monotonic clock, standard error (eprint_str), failure to read a file (code and message of std::io::Error), running a process and its output, exiting in the middle (std::process::exit: the code after it does not run)")]
mod sys {
    #[effects(alloc, io, nondet)]
    fn report(path: &str) -> () {
        match std::fs::read(path) {
            Ok(bytes) => print_u64(bytes.len() as u64),
            Err(e) => {
                print_i64(i64::try_from(e.code).unwrap());
                print_str(e.message.as_str());
                print_str("\n");
            },
        }
    }

    #[effects(alloc, io, nondet)]
    fn main() -> () {
        let args: Vec<String> = std::env::args();
        print_u64(args.len() as u64);
        match std::env::var("BMB_PORT_TEST_UNSET_NAME") {
            None => print_i64(-1),
            Some(v) => print_u64(v.len() as u64),
        }
        match std::env::var("A\0B") {
            None => print_i64(-2),
            Some(v) => print_u64(v.len() as u64),
        }
        let t0: u64 = std::time::monotonic_ms();
        let t1: u64 = std::time::monotonic_ms();
        if t0 <= t1 {
            print_i64(1);
        }
        eprint_str("to standard error\n");
        report("/nonexistent/bmb-port-test");
        let mut argv: Vec<String> = Vec::new();
        argv.push(String::from("-n"));
        argv.push(String::from("out"));
        let input: Vec<u8> = Vec::new();
        match std::process::run("echo", argv.as_slice(), input.as_slice()) {
            Ok(o) => {
                print_i64(i64::try_from(o.status).unwrap());
                print_u64(o.stdout.len() as u64);
                print_u64(o.stderr.len() as u64);
            },
            Err(e) => print_i64(i64::try_from(e.code).unwrap()),
        }
        if args.len() < 100 {
            print_i64(7);
            std::process::exit(0);
        }
        print_i64(8);
    }
}
