#[summary("Several units and an inner unit: definitions come out in path order. main is in the later unit")]
mod zeta {
    #[effects(pure)]
    fn one() -> i64 { 1 }
    #[summary("An inner unit")]
    mod inner {
        #[effects(pure)]
        fn two() -> i64 { 2 }
    }
    #[effects(io)]
    fn main() -> () { }
}

#[summary("The earlier unit: spec functions and proof functions are not emitted")]
mod alpha {
    spec fn pos(x: i64) -> bool { x > 0 }
    proof fn lemma(x: i64)
        requires pos(x),
        ensures pos(x),
    {
    }
    #[effects(pure)]
    fn three() -> i64 { 3 }
    #[effects(pure)]
    fn generic<T: Copy>(x: T) -> i64 { 4 }
}
