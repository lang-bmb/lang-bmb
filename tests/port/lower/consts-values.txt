    const AY: isize = isize::MIN;
    const AZ: usize = usize::MAX;
    struct Pt { x: i64, y: i64 }
    enum Sh { Dot, Rect(i64, i64) }
    struct Wrap<T> { v: T }
    #[effects(pure)]
    fn f() -> [u8; K] { [0; K] }
    #[effects(pure)]
    fn g(a: [i64; K]) -> usize { let b: [u8; K] = [1; K]; K }
    #[summary("An inner unit: written after the lengths that use names of the outer unit")]
