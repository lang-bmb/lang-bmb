#[summary("Two functions of the same name, only one with an Fn bound: the second is read with its own parameters (before the fix stage0 crashed)")]
mod m {
    #[effects(pure)]
    fn plain<F: Fn(i64) -> i64>(f: F, x: i64) -> i64 { f(x) }

    #[effects(pure)]
    fn plain(x: i64) -> i64 { x }

    #[effects(pure)]
    fn twice<F: FnMut(i64) -> i64, G: Fn(i64) -> i64>(f: F, g: G, x: i64) -> i64 { g(f(x)) }

    #[effects(pure)]
    fn twice<T>(x: T) -> T { x }

    #[effects(pure)]
    fn main_like() -> i64 { plain(|y: i64| -> i64 { y }, 1) }
