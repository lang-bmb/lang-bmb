#[summary("aliases")]
mod a {
    #[effects(pure)]
    fn f(n: u64) -> u64
        requires small(n), tiny(n),
    {
        let mut i: u64 = 0;
        let mut h: u64 = 0;
        while i < n
            invariant i <= n, h <= i,
            decreases n - i,
        {
            i = i + 1;
            h = h + 1;
        }
        let mut s: u64 = 0;
        loop
            invariant s <= n, tiny(n),
            decreases n - s,
        {
            if s >= n {
                break;
            }
            s = s + (1 + 0);
        }
        h = h ^ 1;
        h
    }

    spec fn small(n: u64) -> bool {
        n < 1000
    }

    spec fn tiny(n: u64) -> bool {
        n < 100
    }
}
