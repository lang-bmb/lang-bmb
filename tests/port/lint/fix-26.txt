#[summary("c")]
mod m {
    #[effects(pure)]
    fn apply<F: Fn(i64) -> i64>(f: F, x: i64) -> (r: i64)
        requires f.requires(x),
        ensures f.ensures(x, r),
    {
        f(x)
    }

    #[effects(pure)]
    fn c() -> i64 {
        let mut i: usize = 0;
        while i < 3
            invariant i <= 3,
            decreases 3 - i,
        {
            i = i + 1;
        }
        apply::<_>(|x: i64| -> i64 { x.wrapping_add(1) }, 1)
    }
}
