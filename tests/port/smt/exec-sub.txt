    fn signed8(a: i8, b: i8) -> () {
        assert(a + b - b == a) by (bit_vector);
        assert(a * b == b * a) by (bit_vector);
        assert(-(-a) == a) by (bit_vector);
        assert(a - 1 < a) by (bit_vector);
        assert(a + 127 >= -1) by (bit_vector);
        assert(b != 0 ==> a / b * b + a % b == a) by (bit_vector);
        assert(a / 4 * 4 + a % 4 == a) by (bit_vector);
        assert(a / 3 <= 42 && a % 3 < 3) by (bit_vector);
        assert(a / -2 >= -63 && a % -2 > -2) by (bit_vector);
        assert(a / 1 == a && a % 1 == 0) by (bit_vector);
        assert(a & b <= a | b || a < 0 || b < 0) by (bit_vector);
        assert(a ^ a == 0 && !a == -a - 1) by (bit_vector);
        assert(a << 1 >> 1 == a || a < -64 || a > 63) by (bit_vector);
        assert(a >> 7 == 0 || a >> 7 == -1) by (bit_vector);
        assert((a < b || a == b || a > b) && (a <= b || a >= b) && a != b + 1) by (bit_vector);
        assert(i8::MIN <= a && a <= i8::MAX) by (bit_vector);
    }

    #[effects(pure)]
    fn unsigned8(p: u8, q: u8) -> () {
        assert(p + q >= p) by (bit_vector);
        assert(p * q >= 0 && p - q <= 255) by (bit_vector);
        assert(q != 0 ==> p / q * q + p % q == p) by (bit_vector);
        assert(p / 16 == p >> 4 && p % 16 == p & 15) by (bit_vector);
        assert(p / 7 <= 36 && p % 7 <= 6) by (bit_vector);
        assert(p / 1 == p && p % 1 == 0 && p % 2 <= 1) by (bit_vector);
        assert(p & q <= p && p | q >= p && p ^ q <= 255 && !p == 255 - p) by (bit_vector);
        assert(p << 3 >> 3 == p & 31) by (bit_vector);
        assert(p >> q <= p || q >= 8) by (bit_vector);
        assert(p << q >= 0 || q >= 8) by (bit_vector);
        assert(u8::MAX - p >= 0 && u8::MIN <= p) by (bit_vector);
    }

    fn widths(b: i16, c: i32, d: i64, e: isize, q: u16, r: u32, s: u64, t: usize) -> () {

        assert(t - t == 0 && t / 2 <= t) by (bit_vector);
        assert(i64::MIN <= d && d <= i64::MAX && s <= u64::MAX && usize::MAX >= t) by (bit_vector);
        assert(c as i64 + d - d == c as i64 && q as u32 <= 65535) by (bit_vector);

        assert({ let h: i8 = a / 2; let k: i8 = h + h; k == a || k + 1 == a || k - 1 == a }) by (bit_vector);
        assert(o < n ==> !o) by (bit_vector);
        assert(s * s * s * s == s * s * (s * s)) by (bit_vector);

    #[effects(pure)]
    #[uses(low)]
    fn placed(a: &[u32], lim: u32) -> (r: u32) {
        let mut acc: u32 = 0;
        for i in 0..a.len()
            invariant acc <= 255,
        {
            assert(low(a[i]) <= 255 && (acc ^ low(a[i])) <= 255 || acc > 255) by (bit_vector);
            if a[i] < lim {
                assert(a[i] & 1 <= 1) by (bit_vector);
                acc = (acc ^ a[i]) & 255;
            } else {
                proof {
                    assert(lim >> 1 <= lim) by (bit_vector);
                }
            }
        }
        acc
    }

        requires a <= b,
        ensures a & b <= b,

        if a == b {
            assert(a ^ b == 0 ==> a == b) by (bit_vector);
        }
    }
