#[summary("Loops in code: `while` with invariants and `decreases`, `for` over open and closed ranges of every integer type, a `for` starting in the middle of a Vec and one whose Vec changes, `loop` with a `break` value, labelled `break` and `continue`, nested loops, `return` inside loops, loops that assign struct fields, array elements and what is behind `&mut` parameters, a `loop` that never leaves, values with invariants havocked by a loop")]
mod loops {
    struct Pt { x: i64, y: i64 }

    struct Acc { sum: u64, n: usize, cells: [u8; 4] }

    #[invariant(even_ok)]
    struct Even { v: u32 }

    spec fn even_ok(e: &Even) -> bool { e.v % 2 == 0 }

    spec fn le(i: usize, n: usize) -> bool { i <= n }

    spec fn sum_ok(s: u64, i: usize) -> bool { s <= 255 * i }

    spec fn small(a: &[i64]) -> bool {
        forall|k: usize| k < a.len() ==> -1000 < #[trigger] a[k] && a[k] < 1000
    }

    spec fn all_zero(a: &[u8], n: usize) -> bool {
        forall|k: usize| k < n ==> #[trigger] a[k] == 0
    }

    #[effects(pure)]
    fn sum_while(a: &[u8]) -> (r: u64)
        ensures sum_ok(r, a.len()),
    {
        let mut s: u64 = 0;
        let mut i: usize = 0;
        while i < a.len()
            invariant le(i, a.len()), sum_ok(s, i),
            decreases a.len() - i,
        {
            s = s + a[i] as u64;
            i = i + 1;
        }
        s
    }

    #[effects(nondet)]
    fn no_measure(n: u32) -> u32 {
        let mut k: u32 = n;
        loop
            invariant k <= n,
        {
            if k <= 10 {
                break;
            }
            k = k / 2;
        }
        k
    }

    #[effects(pure)]
    fn ranges(n: u8, m: i8, w: u16, x: i16, y: u32, z: i32, p: u64, q: i64, u: usize, v: isize) -> u64 {
        let mut t: u64 = 0;
        for a in 0..n { t = t + 1; }
        for b in m..=100 { t = t + 1; }
        for c in 0..=w { t = t + 2; }
        for d in -5..x { t = t + 1; }
        for e in 1..y { t = t + 1; }
        for g in z..=7 { t = t + 1; }
        for h in 3..p { t = t + 1; }
        for i in q..q { t = t + 1; }
        for j in 0..u { t = t + 1; }
        for k in v..=v { t = t + 1; }
        t
    }

    #[effects(pure)]
    fn empty_and_full(lo: u8) -> u32 {
        let mut t: u32 = 0;
        for a in lo..=255 { t = t + 1; }
        for b in 9..3 { t = t + 100; }
        for c in 0..=0 { t = t + 1; }
        t
    }

    #[effects(alloc)]
    fn mid_vec(v: &Vec<u8>, from: usize) -> u64
        requires from <= v.len(),
    {
        let mut s: u64 = 0;
        for i in from..v.len()
            invariant s <= 255 * (i - from),
        {
            s = s + v[i] as u64;
        }
        s
    }

    #[effects(alloc)]
    fn grows(n: usize) -> Vec<usize>
        requires n < 1000,
    {
        let mut out: Vec<usize> = Vec::new();
        for i in 0..n
            invariant out.len() == i,
        {
            out.push(i);
        }
        let old_len: usize = out.len();
        for j in old_len..old_len + 3
            invariant out.len() == j,
        {
            out.push(j * 2);
        }
        out
    }

    #[effects(alloc)]
    fn shrinks(v: &mut Vec<i64>) -> i64 {
        let mut s: i64 = 0;
        while v.len() > 0
            invariant -1000 < s && s < 1000,
            decreases v.len(),
        {
            match v.pop() {
                None => {},
                Some(x) => { if x > 0 && x < 10 && s < 900 { s = s + x; } },
            }
        }
        s
    }

    #[effects(pure)]
    fn break_value(n: u64) -> u64
        requires n < 1000000,
    {
        let mut k: u64 = 0;
        let r: u64 = loop
            invariant k <= 1000,
            decreases 1001 - k,
        {
            if k * k >= n {
                break k;
            }
            if k == 1000 {
                break 0;
            }
            k = k + 1;
        };
        r
    }

    #[effects(pure)]
    fn break_struct(a: &[i64]) -> Pt {
        let mut i: usize = 0;
        let p: Pt = loop
            invariant i <= a.len(),
            decreases a.len() - i,
        {
            if i == a.len() {
                break Pt { x: 0, y: 0 };
            }
            if a[i] < 0 {
                break Pt { x: a[i], y: 1 };
            }
            i = i + 1;
        };
        p
    }

    #[effects(nondet)]
    fn never() -> u8 {
        let mut k: u8 = 0;
        loop {
            k = k ^ 1;
        }
    }

    #[effects(pure)]
    fn labelled(a: &[i64], t: i64) -> i64
        requires small(a),
    {
        let mut found: i64 = -1;
        let mut i: usize = 0;
        'outer: while i < a.len()
            invariant i <= a.len(),
            decreases a.len() - i,
        {
            let mut j: usize = i + 1;
            while j < a.len()
                invariant j <= a.len(),
                decreases a.len() - j,
            {
                if a[j] == 0 {
                    i = i + 1;
                    continue 'outer;
                }
                if a[i] + a[j] == t {
                    found = a[i];
                    break 'outer;
                }
                j = j + 1;
            }
            i = i + 1;
        }
        found
    }

    #[effects(pure)]
    fn labelled_for(n: usize, m: usize) -> usize {
        let mut c: usize = 0;
        'rows: for i in 0..n
            invariant c <= i,
        {
            for j in 0..=m {
                if j == i {
                    continue 'rows;
                }
                if j > 100 {
                    break 'rows;
                }
                if j % 2 == 0 {
                    continue;
                }
            }
            c = c + 1;
        }
        c
    }

    #[effects(pure)]
    fn labelled_loop(n: u32) -> u32 {
        let mut k: u32 = 0;
        let r: u32 = 'top: loop
            invariant k <= n,
            decreases n - k,
        {
            let mut d: u32 = 0;
            loop
                invariant d <= 3,
                decreases 3 - d,
            {
                if d == 3 {
                    break;
                }
                if k == n {
                    break 'top k;
                }
                d = d + 1;
            }
            k = k + 1;
        };
        r
    }

    #[effects(pure)]
    fn returns(a: &[u8], x: u8) -> (r: Option<usize>) {
        for i in 0..a.len() {
            if a[i] == x {
                return Some(i);
            }
        }
        let mut j: usize = 0;
        while j < 4
            invariant j <= 4,
            decreases 4 - j,
        {
            if j == 3 {
                return None;
            }
            j = j + 1;
        }
        loop
            decreases 0,
        {
            return Some(0);
        }
    }

    #[effects(pure)]
    fn fields(acc: &mut Acc, a: &[u8]) -> ()
        requires acc.n == 0 && acc.sum == 0,
    {
        for i in 0..a.len()
            invariant acc.n == i && sum_ok(acc.sum, i),
        {
            acc.sum = acc.sum + a[i] as u64;
            acc.n = acc.n + 1;
            if i < 4 {
                acc.cells[i] = a[i];
            }
        }
    }

    #[effects(pure)]
    fn elements(a: &mut [u8], p: &mut Pt, total: &mut i64) -> ()
        ensures all_zero(a, a.len()),
    {
        let mut local: [i64; 3] = [1, 2, 3];
        let mut q: Pt = Pt { x: 0, y: 0 };
        for i in 0..a.len()
            invariant all_zero(a, i),
        {
            a[i] = 0;
            local[i % 3] = 0;
            q.x = q.y;
            p.y = 0;
            *total = 0;
        }
    }

    #[effects(pure)]
    fn clear(a: &mut [u8]) -> () {
        for i in 0..a.len() {
            a[i] = 0;
        }
    }

    #[effects(pure)]
    fn passes_on(a: &mut [u8], b: &[u8], n: u32) -> () {
        let mut k: u32 = 0;
        while k < n
            invariant k <= n,
            decreases n - k,
        {
            clear(a);
            let m: usize = b.len();
            k = k + 1;
        }
    }

    #[effects(pure)]
    fn swaps(a: &mut [Pt], n: usize) -> ()
        requires n <= a.len(),
    {
        let mut hold: Pt = Pt { x: 0, y: 0 };
        for i in 0..n {
            let was: Pt = std::mem::replace(&mut a[i], Pt { x: 1, y: 1 });
            hold = was;
        }
    }

    #[effects(pure)]
    fn keeps_valid(e: Even, n: u32) -> Even {
        let mut cur: Even = e;
        let mut i: u32 = 0;
        while i < n
            invariant i <= n,
            decreases n - i,
        {
            if cur.v < 100 {
                cur = Even { v: cur.v + 2 };
            }
            i = i + 1;
        }
        cur
    }

    #[effects(pure)]
    fn valid_param(e: &mut Even, n: u32) -> () {
        for i in 0..n {
            if i == 7 {
                continue;
            }
            e.v = 4;
        }
    }

    #[effects(pure)]
    fn in_condition(a: &[u8]) -> usize {
        let mut i: usize = 0;
        while i < a.len() && a[i] != 0
            invariant i <= a.len(),
            decreases a.len() - i,
        {
            i = i + 1;
        }
        let mut t: usize = 0;
        if i > 2 {
            for j in 0..i { t = t + 1; }
        } else {
            for j in i..3 { t = t + 2; }
        }
        t
    }

    #[effects(pure)]
    fn loop_match(o: &mut Option<u32>, n: u32) -> u32 {
        let mut t: u32 = 0;
        for i in 0..n {
            match &mut *o {
                None => { t = 1; },
                Some(x) => { *x = i; },
            }
        }
        t
    }
}
