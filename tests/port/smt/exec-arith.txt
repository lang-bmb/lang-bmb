#[summary("Arithmetic in code: the overflow obligations of + - * and negation at each width and sign, the facts of a product (its sign and zero, and its bound from the upper bounds the path states for the operands, for a square too), division and remainder (the divisor, MIN % -1, the lemmas of the truncating and of the unsigned forms), shifts by constant and variable amounts of signed and unsigned types, bit operations (linear forms, named bit-vectors and their facts, bool), conversions (try_from(..).unwrap(), .ok(), as, f64::from), the built-in methods (wrapping, saturating and checked arithmetic, bit methods with constant and variable amounts, the bytes of an integer), f64 arithmetic and methods, short-circuit && and || with a call on the right")]
mod arith {
    spec fn below(a: u64, n: u64) -> bool { a < n }

    #[effects(pure)]
    fn widths(a: i8, b: i16, c: i32, d: i64, e: isize, f: u8, g: u16, h: u32, i: u64, j: usize) -> (r: i64) {
        let a2: i8 = a + 1;
        let b2: i16 = b - 2;
        let c2: i32 = c * 3;
        let d2: i64 = -d;
        let e2: isize = e + e;
        let f2: u8 = f + f;
        let g2: u16 = g - 1;
        let h2: u32 = h * h;
        let i2: u64 = i + 18446744073709551615;
        let j2: usize = j - j;
        let lit: i64 = -9223372036854775808;
        d2 + lit
    }

    #[effects(pure)]
    fn products(a: u64, b: u64, c: i32, n: u32) -> (r: u64)
        requires a < 1000 && b <= 2000 && n < 60000,
    {
        let p: u64 = a * b;
        let sq: u32 = n * n;
        let k: u64 = a * 7;
        let s: i32 = c * c;
        if a < 10 { p * a } else { k }
    }

    #[effects(pure)]
    fn division(a: i64, b: i64, c: u32, d: u32, e: i8) -> (r: i64)
        requires b != 0,
    {
        let q: i64 = a / b;
        let m: i64 = a % b;
        let uq: u32 = c / d;
        let um: u32 = c % d;
        let k: i64 = a / 4 + a % -3;
        let e2: i8 = e / -1;
        let z: u32 = c / 0;
        q + m
    }

    #[effects(pure)]
    fn shifts(a: u32, b: i64, n: u32, s: i32, w: u8, big: u64) -> (r: u32) {
        let x: u32 = a << 3;
        let y: u32 = a >> 31;
        let over: u32 = a << 32;
        let neg: i64 = b >> 2;
        let var: u32 = a << n;
        let signed: u32 = a >> s;
        let wide: i64 = b << big;
        let narrow: u8 = w >> w;
        let lit: u32 = 1 << 5;
        x + y
    }

    #[effects(pure)]
    fn bits(a: u32, b: u32, c: i16, d: i16, p: bool, q: bool, e: u8) -> (r: u32) {
        let m: u32 = a & 255;
        let x: u32 = a & b;
        let y: u32 = a | b;
        let z: u32 = a ^ 4294967295;
        let n: u32 = !a;
        let sc: i16 = c & d;
        let sn: i16 = !c;
        let so: i16 = c | -1;
        let pb: bool = p & q | !p ^ q;
        let lit: u8 = e ^ 0x0f;
        m + (x & 1)
    }

    #[effects(pure)]
    fn convert(a: i64, b: u8, c: u32, d: usize, e: u64, x: f64) -> (r: f64) {
        let n: u8 = u8::try_from(a).unwrap();
        let w: i64 = i64::try_from(b).unwrap();
        let same: u64 = u64::try_from(d).unwrap();
        let o: Option<i16> = i16::try_from(c).ok();
        let cast: i32 = e as i32;
        let wide: u64 = c as u64;
        let f: f64 = f64::from(c);
        let g: f64 = f64::from(a);
        let h: f64 = f64::from(e);
        f + g + f64::from(x)
    }

    #[effects(pure)]
    fn methods(a: u32, b: u32, c: i64, d: i8, n: u32) -> (r: u32) {
        let w: u32 = a.wrapping_add(b).wrapping_sub(1).wrapping_mul(b);
        let s: i64 = c.saturating_add(c).saturating_sub(1).saturating_mul(c);
        let ca: Option<u32> = a.checked_add(b);
        let cs: Option<i8> = d.checked_sub(1);
        let cm: Option<i64> = c.checked_mul(c);
        let cd: Option<i64> = c.checked_div(c);
        let cr: Option<u32> = a.checked_rem(b);
        let sd: i8 = d.saturating_mul(d).wrapping_sub(d);
        w
    }

    #[effects(pure)]
    fn bit_methods(a: u32, b: u8, c: i64, n: u32) -> (r: u32) {
        let ones: u32 = a.count_ones() + a.count_zeros() + a.leading_zeros() + a.trailing_zeros();
        let more: u32 = b.leading_ones() + c.trailing_ones();
        let rot: u32 = a.rotate_left(3) ^ a.rotate_right(n);
        let sw: u32 = a.swap_bytes() ^ a.reverse_bits();
        let sb: u8 = b.swap_bytes();
        let ws: u32 = a.wrapping_shl(n) ^ a.wrapping_shr(7);
        let cs: Option<u32> = a.checked_shl(n);
        let cr: Option<i64> = c.checked_shr(65);
        let p: bool = a.is_power_of_two();
        ones
    }

    #[effects(pure)]
    fn bytes(a: u32, b: i16, bs: [u8; 4]) -> (r: u32) {
        let be: [u8; 4] = a.to_be_bytes();
        let le: [u8; 2] = b.to_le_bytes();
        let back: u32 = u32::from_be_bytes(bs);
        let neg: i16 = i16::from_le_bytes(le);
        back ^ u32::from_le_bytes(be)
    }

    #[effects(pure)]
    fn floats(a: f64, b: f64, n: i32) -> (r: bool) {
        let s: f64 = a + b - a * b / a;
        let m: f64 = -a;
        let z: f64 = -0.0;
        let big: f64 = f64::MAX + f64::INFINITY - f64::EPSILON;
        let t: f64 = a.sqrt().abs().min(b).max(1.5);
        let c: bool = a < b || a <= b && a > b || a >= b || a == b || a != b;
        c && a.is_nan() || s.is_finite() && !m.is_finite()
    }

    #[effects(pure)]
    fn id(a: u64) -> (r: u64)
        requires below(a, 10),
        ensures below(r, 10),
    {
        a
    }

    #[effects(pure)]
    fn lazy(a: u64, b: u64) -> (r: bool) {
        let x: bool = a < 10 && id(a) < 5;
        let y: bool = b >= 10 || id(b) == b / (10 - b);
        let z: bool = a < 3 && b < 3 && a + b < 6 || a > 100;
        x || y && z
    }
}
