#[summary("t")]
mod ns {
    #[effects(pure)]
    fn grid(n: u64) -> (r: u64)
        requires n < 100,
        ensures r <= n * n,
    {
        let mut s: u64 = 0;
        let mut i: u64 = 0;
        while i < n
            invariant i <= n, s <= n * i,
            decreases n - i,
        {
            let mut j: u64 = 0;
            while j < n
                decreases n - j,
            {
                s = s + 1;
                j = j + 1;
            }
            i = i + 1;
        }
        s
    }
}
